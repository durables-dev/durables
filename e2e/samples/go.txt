package reports

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/hex"
	"os"

	"github.com/aws/aws-sdk-go-v2/aws"
	"github.com/aws/aws-sdk-go-v2/config"
	"github.com/aws/aws-sdk-go-v2/service/s3"
	"go.temporal.io/sdk/activity"
	"go.temporal.io/sdk/temporal"
)

type ArchiveRequest struct {
	ReportID string
	CSV      string
}

type ArchivedReport struct {
	Bucket string
	Key    string
	SHA256 string
}

func ArchiveReport(ctx context.Context, request ArchiveRequest) (ArchivedReport, error) {
	bucket := os.Getenv("ARCHIVE_BUCKET")
	if bucket == "" {
		return ArchivedReport{}, temporal.NewNonRetryableApplicationError(
			"ARCHIVE_BUCKET is not set",
			"MissingConfiguration",
			nil,
		)
	}

	cfg, err := config.LoadDefaultConfig(ctx)
	if err != nil {
		return ArchivedReport{}, err
	}
	client := s3.NewFromConfig(cfg)

	key := "reports/" + request.ReportID + ".csv"
	activity.RecordHeartbeat(ctx, key)

	sum := sha256.Sum256([]byte(request.CSV))
	_, err = client.PutObject(ctx, &s3.PutObjectInput{
		Bucket: aws.String(bucket),
		Key:    aws.String(key),
		Body:   bytes.NewReader([]byte(request.CSV)),
	})
	if err != nil {
		return ArchivedReport{}, err
	}

	return ArchivedReport{Bucket: bucket, Key: key, SHA256: hex.EncodeToString(sum[:])}, nil
}
