import { createHash } from "node:crypto";
import {
  PutObjectCommand,
  S3Client,
  S3ServiceException,
} from "@aws-sdk/client-s3";
import { heartbeat } from "@temporalio/activity";
import { ApplicationFailure } from "@temporalio/common";

export interface ReportUpload {
  reportId: string;
  csv: string;
}

export interface StoredReport {
  bucket: string;
  key: string;
  sha256: string;
}

const s3 = new S3Client({ region: process.env.AWS_REGION });

export async function uploadReport(upload: ReportUpload): Promise<StoredReport> {
  const bucket = process.env["REPORTS_BUCKET"];
  if (!bucket) {
    throw ApplicationFailure.nonRetryable(
      "REPORTS_BUCKET is not set",
      "MissingConfiguration",
    );
  }
  const key = `reports/${upload.reportId}.csv`;
  const sha256 = createHash("sha256").update(upload.csv).digest("hex");

  heartbeat({ key });
  try {
    await s3.send(
      new PutObjectCommand({
        Bucket: bucket,
        Key: key,
        Body: upload.csv,
        ContentType: "text/csv",
      }),
    );
  } catch (err) {
    if (err instanceof S3ServiceException && err.name === "NoSuchBucket") {
      throw ApplicationFailure.nonRetryable(
        `Bucket ${bucket} does not exist`,
        "NoSuchBucket",
      );
    }
    throw err;
  }
  return { bucket, key, sha256 };
}
