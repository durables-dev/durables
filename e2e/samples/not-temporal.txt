"""Count the most common email domains in a CSV export."""

import csv
import sys
from collections import Counter


def top_domains(path: str, limit: int = 10) -> list[tuple[str, int]]:
    counts: Counter[str] = Counter()
    with open(path, newline="", encoding="utf-8") as export:
        for row in csv.DictReader(export):
            email = row.get("email", "")
            if "@" in email:
                counts[email.rsplit("@", 1)[1].lower()] += 1
    return counts.most_common(limit)


if __name__ == "__main__":
    for domain, count in top_domains(sys.argv[1]):
        print(f"{count:>6}  {domain}")
