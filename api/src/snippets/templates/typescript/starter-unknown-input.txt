import { Client, Connection } from "@temporalio/client";

import { {{workflowFunction}} } from "./workflows";

async function run() {
    const connection = await Connection.connect({ address: "localhost:7233" });
    const client = new Client({ connection });
    // Fill in the arguments {{activity}} takes, in order.
    const args = [] as unknown as Parameters<typeof {{workflowFunction}}>;
    const result = await client.workflow.execute({{workflowFunction}}, {
        args,
        taskQueue: {{taskQueue}},
        workflowId: {{workflowId}},
    });
    console.log(result);
}

run().catch((err: unknown) => {
    console.error(err);
    process.exit(1);
});
